# Chat App Implementation Plan (Flutter)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

> **API contract:** See `docs/superpowers/plans/2026-03-25-chat-api-contract.md` for all expected backend endpoints, request/response formats, and infrastructure requirements. This plan assumes the API is available.

**Goal:** Build the Flutter chat module — local persistence, real-time SSE streaming UI, session management, and feedback — consuming the chat API contract.

**Architecture:** Server-first for messages (requires internet), local Drift cache for instant history. SSE streaming via Dio `ResponseType.stream`. Feedback uses existing SyncQueue for offline resilience. Riverpod for DI and reactive streams.

**Tech Stack:** Flutter/Dart, Drift (SQLite), Riverpod, Dio. Backend uses OpenRouter (OpenAI-compatible) for model flexibility.

```dart
class ChatRemoteDataSource {
  final ApiClient _apiClient;
  ChatRemoteDataSource(this._apiClient);

  Future<ApiResult<List<Map<String, dynamic>>>> fetchSessions({String? since}) { ... }
  Future<ApiResult<Map<String, dynamic>>> createSession({required String persona, String title = ''}) { ... }
  Future<ApiResult<void>> deleteSession(int sessionId) { ... }

  /// POST message → returns { user_message: {...}, assistant_message: {id, ...} }
  Future<ApiResult<Map<String, dynamic>>> sendMessage(int sessionId, String content) { ... }

  /// SSE stream — opens GET connection, yields token strings.
  Stream<String> streamResponse(int sessionId, int assistantMessageId) async* {
    final response = await _apiClient.dio.get(
      ApiEndpoints.chatStream(sessionId),
      queryParameters: {'message_id': assistantMessageId},
      options: Options(
        responseType: ResponseType.stream,
        headers: {'Accept': 'text/event-stream'},
      ),
    );

    final stream = response.data.stream as Stream<List<int>>;
    String buffer = '';

    await for (final chunk in stream.transform(utf8.decoder)) {
      buffer += chunk;
      while (buffer.contains('\n\n')) {
        final eventEnd = buffer.indexOf('\n\n');
        final event = buffer.substring(0, eventEnd);
        buffer = buffer.substring(eventEnd + 2);

        for (final line in event.split('\n')) {
          if (line.startsWith('data: ')) {
            final data = line.substring(6);
            if (data == '[DONE]') return;
            // Parse JSON to extract token or error
            try {
              final json = jsonDecode(data) as Map<String, dynamic>;
              if (json.containsKey('error')) {
                throw Exception(json['error']);
              }
              yield json['token'] as String;
            } catch (e) {
              if (e is Exception) rethrow;
              yield data; // fallback: yield raw
            }
          }
        }
      }
    }
  }

  Future<ApiResult<List<Map<String, dynamic>>>> fetchAllMessages({String? since}) { ... }
  Future<ApiResult<Map<String, dynamic>>> fetchMessages(int sessionId, {String? since}) { ... }
  Future<ApiResult<void>> updateFeedback(int messageId, {String? type, String? detail}) { ... }
}
```

```dart
class ChatRepository {
  final ChatLocalDataSource _local;
  final ChatRemoteDataSource _remote;
  final ConnectivityService _connectivity;
  final SyncQueue _syncQueue;

  // Track active SSE streams for cancellation
  final Map<String, StreamSubscription> _activeStreams = {};

  ChatRepository({
    required ChatLocalDataSource localDataSource,
    required ChatRemoteDataSource remoteDataSource,
    required ConnectivityService connectivity,
    required SyncQueue syncQueue,
  }) : _local = localDataSource,
       _remote = remoteDataSource,
       _connectivity = connectivity,
       _syncQueue = syncQueue;

  // === Reactive streams (from local cache) ===
  Stream<List<ChatSessionsTableData>> watchSessions(String userId) => _local.watchSessions(userId);
  Stream<List<ChatMessagesTableData>> watchMessages(String sessionId) => _local.watchMessages(sessionId);

  // === Create session (server-first) ===
  // Throws if offline — UI shows "requires internet"
  Future<ChatSessionsTableData> createSession(String userId, ChatPersona persona) async {
    if (!await _connectivity.isOnline) throw Exception('Requires internet');

    final result = await _remote.createSession(persona: persona.id);
    if (result.isFailure) throw Exception(result.error);

    final serverData = result.data!['data'] ?? result.data!;
    final localId = const Uuid().v4();

    final companion = ChatSessionsTableCompanion(
      id: Value(localId),
      userId: Value(userId),
      serverId: Value(serverData['id'] as int),
      title: Value(serverData['title'] as String? ?? ''),
      persona: Value(persona.id),
      syncStatus: const Value('synced'),
    );
    await _local.insertSession(companion);
    return (await _local.getSessionById(localId))!;
  }

  // === Send message (server-first + SSE) ===
  // Returns stream of ChatStreamEvent. Throws if offline.
  Stream<ChatStreamEvent> sendMessage(String sessionId, String content) async* {
    if (!await _connectivity.isOnline) throw Exception('Requires internet');

    final session = await _local.getSessionById(sessionId);
    if (session == null || session.serverId == null) throw Exception('Session not found');

    // Step 1: POST message → get both IDs
    final result = await _remote.sendMessage(session.serverId!, content);
    if (result.isFailure) {
      yield ChatStreamError(result.error ?? 'Failed to send');
      return;
    }

    final data = result.data!['data'] ?? result.data!;
    final userMsgData = data['user_message'] as Map<String, dynamic>;
    final assistantMsgData = data['assistant_message'] as Map<String, dynamic>;

    // Insert user message locally
    await _local.insertMessage(ChatMessagesTableCompanion(
      id: Value(const Uuid().v4()),
      sessionId: Value(sessionId),
      serverId: Value(userMsgData['id'] as int),
      role: const Value('user'),
      content: Value(content),
      syncStatus: const Value('synced'),
    ));

    // Insert empty assistant message locally (placeholder)
    final assistantLocalId = const Uuid().v4();
    await _local.insertMessage(ChatMessagesTableCompanion(
      id: Value(assistantLocalId),
      sessionId: Value(sessionId),
      serverId: Value(assistantMsgData['id'] as int),
      role: const Value('assistant'),
      content: const Value(''),
      syncStatus: const Value('synced'),
    ));

    // Update session lastMessageAt
    await _local.updateSession(sessionId, ChatSessionsTableCompanion(
      lastMessageAt: Value(DateTime.now()),
    ));

    // Step 2: Open SSE stream
    String fullContent = '';
    try {
      await for (final token in _remote.streamResponse(
        session.serverId!,
        assistantMsgData['id'] as int,
      )) {
        fullContent += token;
        // Update local message content incrementally
        await _local.updateMessageContent(assistantLocalId, fullContent);
        yield ChatStreamToken(token);
      }
      yield ChatStreamDone(fullContent, assistantMsgData['id'] as int);
    } catch (e) {
      // Save whatever we got so far
      if (fullContent.isNotEmpty) {
        await _local.updateMessageContent(assistantLocalId, fullContent);
      }
      yield ChatStreamError(e.toString());
    }
  }

  // === Feedback (SyncQueue for offline resilience) ===
  Future<void> submitFeedback(String messageId, String? type, String? detail) async {
    await _local.updateMessageFeedback(messageId, type, detail);

    final message = await _local.getMessageByServerId(/* need serverId */);
    // Look up serverId from local message, enqueue UPDATE
    await _syncQueue.enqueue(
      tableName: 'chat_messages',
      recordId: messageId,
      operation: 'UPDATE',
      payload: {'feedback_type': type, 'feedback_detail': detail},
    );
  }

  // === Delete session ===
  Future<void> deleteSession(String sessionId) async {
    final session = await _local.getSessionById(sessionId);
    if (session?.serverId != null && await _connectivity.isOnline) {
      await _remote.deleteSession(session!.serverId!);
    } else if (session?.serverId != null) {
      await _syncQueue.enqueue(
        tableName: 'chat_sessions',
        recordId: sessionId,
        operation: 'DELETE',
        payload: {'server_id': session!.serverId},
      );
    }
    await _local.deleteSession(sessionId);
  }

  // === Pull history (called by SyncService) ===
  Future<void> pullHistory(String userId, {String? since}) async {
    // Pull sessions
    final sessResult = await _remote.fetchSessions(since: since);
    if (sessResult.isSuccess && sessResult.data != null) {
      for (final s in sessResult.data!) {
        final existing = await _local.getSessionByServerId(s['id'] as int);
        if (existing == null) {
          await _local.insertSession(ChatSessionsTableCompanion(
            id: Value(const Uuid().v4()),
            userId: Value(userId),
            serverId: Value(s['id'] as int),
            title: Value(s['title'] as String? ?? ''),
            persona: Value(s['persona'] as String? ?? 'ai_teacher'),
            lastMessageAt: Value(DateTime.parse(s['last_message_at'])),
            syncStatus: const Value('synced'),
          ));
        }
      }
    }

    // Pull messages (single endpoint, all sessions)
    final msgResult = await _remote.fetchAllMessages(since: since);
    if (msgResult.isSuccess && msgResult.data != null) {
      for (final m in msgResult.data!) {
        final existing = await _local.getMessageByServerId(m['id'] as int);
        if (existing == null) {
          // Find local session by server session ID
          final localSession = await _local.getSessionByServerId(m['chat_session_id'] as int);
          if (localSession == null) continue; // orphan message, skip

          await _local.insertMessage(ChatMessagesTableCompanion(
            id: Value(const Uuid().v4()),
            sessionId: Value(localSession.id),
            serverId: Value(m['id'] as int),
            role: Value(m['role'] as String),
            content: Value(m['content'] as String? ?? ''),
            messageType: Value(m['message_type'] as String? ?? 'text'),
            feedbackType: Value(m['feedback_type'] as String?),
            syncStatus: const Value('synced'),
          ));
        } else {
          // Only update feedback if server has newer data
          if (m['feedback_type'] != null && existing.feedbackType == null) {
            await _local.updateMessageFeedback(
              existing.id,
              m['feedback_type'] as String?,
              m['feedback_detail'] as String?,
            );
          }
        }
      }
    }
  }

  void cancelStream(String sessionId) {
    _activeStreams[sessionId]?.cancel();
    _activeStreams.remove(sessionId);
  }

  void dispose() {
    for (final sub in _activeStreams.values) { sub.cancel(); }
    _activeStreams.clear();
  }
}
```
